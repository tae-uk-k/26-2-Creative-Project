## 2. `seelo/db.py`

```python
from collections.abc import AsyncIterator
from contextlib import asynccontextmanager
from sqlalchemy.ext.asyncio import AsyncEngine, AsyncSession, async_sessionmaker, create_async_engine
from seelo.config import get_settings

_engine: AsyncEngine | None = None
_factory: async_sessionmaker[AsyncSession] | None = None


def get_engine() -> AsyncEngine:
    global _engine, _factory
    if _engine is None:
        _engine = create_async_engine(get_settings().database_url, pool_pre_ping=True,
                                      pool_size=10, max_overflow=5)
        _factory = async_sessionmaker(_engine, expire_on_commit=False)
    return _engine


def session_factory() -> async_sessionmaker[AsyncSession]:
    get_engine()
    assert _factory is not None
    return _factory


async def dispose_engine() -> None:
    global _engine, _factory
    if _engine is not None:
        await _engine.dispose()
    _engine, _factory = None, None


async def get_db() -> AsyncIterator[AsyncSession]:
    """FastAPI 의존성. 커밋은 서비스 함수가 직접 한다. 남은 트랜잭션은 버린다."""
    async with session_factory()() as session:
        try:
            yield session
        finally:
            if session.in_transaction():
                await session.rollback()


@asynccontextmanager
async def session_scope() -> AsyncIterator[AsyncSession]:
    """작업(worker)용. 블록 안에서 필요한 곳마다 await session.commit()."""
    async with session_factory()() as session:
        try:
            yield session
        finally:
            if session.in_transaction():
                await session.rollback()
```

```python
class AppError(Exception):
    def __init__(self, code: str, details: dict[str, Any] | None = None) -> None:
        if code not in ERRORS:
            raise ValueError(f"unknown error code {code}")
        self.code = code
        self.status, template = ERRORS[code]
        self.details = details or {}
        provider = self.details.get("provider")
        self.message = template.format(provider=PROVIDER_LABEL.get(provider, provider or ""))
        super().__init__(code)


def _body(code: str, message: str, details: dict[str, Any]) -> dict[str, Any]:
    return {"error": {"code": code, "message": message, "details": details}}


# Pydantic 오류 type → 04 2장 필드 코드
FIELD_CODE = {"missing": "required", "string_too_long": "too_long", "string_too_short": "required",
              "too_long": "too_long", "bool_type": "required", "literal_error": "invalid"}
# 모델 검사(model_validator)에서 낸 오류의 필드 이름(camelCase)
CUSTOM_FIELD = {"password_mismatch": "passwordConfirm", "terms_required": "agreeTerms"}


def install_handlers(app: FastAPI) -> None:
    @app.exception_handler(AppError)
    async def _app_error(_: Request, e: AppError) -> JSONResponse:
        return JSONResponse(_body(e.code, e.message, e.details), status_code=e.status)

    @app.exception_handler(RequestValidationError)
    async def _validation(_: Request, e: RequestValidationError) -> JSONResponse:
        fields: dict[str, str] = {}
        for err in e.errors():
            loc = [str(x) for x in err["loc"] if x not in ("body", "query", "path")]
            name = loc[-1] if loc else "_"
            t = err["type"]
            if t in CUSTOM_FIELD:                          # 모델 단위 검사는 loc이 비어 있어 필드를 정해 준다
                name = CUSTOM_FIELD[t]
            if t in ("invalid_email", "weak_password", "password_mismatch", "terms_required"):
                code = t                                   # PydanticCustomError(type=...)로 직접 낸 코드
            elif name.lower().endswith("email") and t == "value_error":
                code = "invalid_email"
            else:
                code = FIELD_CODE.get(t, "invalid")
            fields.setdefault(name, code)
        status, msg = ERRORS["VALIDATION_ERROR"]
        return JSONResponse(_body("VALIDATION_ERROR", msg, {"fields": fields}), status_code=status)

    @app.exception_handler(Exception)
    async def _unhandled(_: Request, e: Exception) -> JSONResponse:
        log.exception("unhandled_error", error_type=type(e).__name__)
        status, msg = ERRORS["INTERNAL_ERROR"]
        return JSONResponse(_body("INTERNAL_ERROR", msg, {}), status_code=status)
```

```python
from datetime import UTC, date, datetime, timedelta
from zoneinfo import ZoneInfo

KST = ZoneInfo("Asia/Seoul")


def now_utc() -> datetime:
    return datetime.now(UTC)


def kst_day_range(at: datetime | None = None) -> tuple[datetime, datetime]:
    """at이 속한 한국 날짜의 [00:00, 다음날 00:00)을 UTC로."""
    at = at or now_utc()
    d = at.astimezone(KST).date()
    start = datetime(d.year, d.month, d.day, tzinfo=KST)
    return start.astimezone(UTC), (start + timedelta(days=1)).astimezone(UTC)


def kst_date(at: datetime) -> date:
    return at.astimezone(KST).date()
```

## 5. `seelo/logging.py`

```python
import logging, re, sys
import structlog

SENSITIVE_KEYS = ("password", "token", "secret", "authorization", "cookie", "email", "api_key", "code")
EMAIL_RE = re.compile(r"([^@\s\"'<>]+)@([^@\s\"'<>]+)")


_NEVER_MASK = {"event", "level", "timestamp", "logger"}


def _sensitive(key, value) -> bool:
    k = str(key).lower()
    if k.endswith("_id") and isinstance(value, int) and not isinstance(value, bool):
        return False                      # email_log_id=3 같은 행 번호는 가리지 않는다
    return any(s in k for s in SENSITIVE_KEYS)


def _mask(value):
    if isinstance(value, str):
        return EMAIL_RE.sub(lambda m: "***@" + m.group(2), value)
    if isinstance(value, dict):
        return {k: ("[REDACTED]" if _sensitive(k, v) else _mask(v)) for k, v in value.items()}
    if isinstance(value, (list, tuple)):
        return [_mask(v) for v in value]
    return value


def mask_text(value: str) -> str:
    """오류 메시지를 DB(email_logs.error 등)·로그에 남길 때 이메일 주소를 가린다."""
    return EMAIL_RE.sub(lambda m: "***@" + m.group(2), value)


def redact(_, __, event_dict):
    return {k: (v if k in _NEVER_MASK else "[REDACTED]" if _sensitive(k, v) else _mask(v))
            for k, v in event_dict.items()}


def setup_logging(level: str = "INFO") -> None:
    logging.basicConfig(stream=sys.stdout, level=level, format="%(message)s")
    structlog.configure(
        processors=[
            structlog.contextvars.merge_contextvars,
            structlog.processors.add_log_level,
            structlog.processors.TimeStamper(fmt="iso", utc=True),
            structlog.processors.format_exc_info,
            redact,
            structlog.processors.JSONRenderer(ensure_ascii=False),
        ],
        wrapper_class=structlog.make_filtering_bound_logger(logging.getLevelName(level)),
    )
```

- 키 `event`(로그 이름)는 가리지 않도록 `SENSITIVE_KEYS`에 넣지 않는다. 이메일 형태가 든 이벤트 이름을 쓰지 않는다.
- 테스트: `{"email": "a@b.com", "note": "user a@b.com", "token": "x"}` → `{"email": "[REDACTED]", "note": "user ***@b.com", "token": "[REDACTED]"}`. `{"email_log_id": 3}`은 그대로, `{"email_id": "a@b.com"}`은 가림.
- 개수를 세는 키 이름에 `token`·`email`을 넣지 않는다(예: 정리 작업 결과는 `auth_links`).

## 6. `seelo/main.py`

```python
from contextlib import asynccontextmanager
import uuid
import sentry_sdk
import structlog
from fastapi import FastAPI, Request
from starlette.middleware.sessions import SessionMiddleware
from seelo.api.router import api_router
from seelo.config import get_settings
from seelo.db import dispose_engine
from seelo.errors import install_handlers
from seelo.logging import setup_logging
from seelo.worker.app import app as queue_app


def create_app() -> FastAPI:
    s = get_settings()
    setup_logging(s.log_level)
    if s.sentry_dsn:
        sentry_sdk.init(dsn=s.sentry_dsn, send_default_pii=False, traces_sample_rate=0.0)

    @asynccontextmanager
    async def lifespan(_: FastAPI):
        async with queue_app.open_async():
            yield
        await dispose_engine()

    app = FastAPI(title="seelo API", lifespan=lifespan,
                  docs_url="/api/docs" if s.app_env != "prod" else None,
                  redoc_url=None, openapi_url="/api/openapi.json" if s.app_env != "prod" else None)

    @app.middleware("http")
    async def request_id(request: Request, call_next):
        rid = request.headers.get("X-Request-Id") or uuid.uuid4().hex
        structlog.contextvars.bind_contextvars(request_id=rid, path=request.url.path)
        try:
            response = await call_next(request)
        finally:
            structlog.contextvars.clear_contextvars()
        response.headers["X-Request-Id"] = rid
        return response

    app.add_middleware(SessionMiddleware, secret_key=s.secret_key.get_secret_value(),
                       session_cookie="seelo_oauth", max_age=600, same_site="lax",
                       https_only=s.cookie_secure)
    install_handlers(app)
    app.include_router(api_router, prefix="/api")
    return app


app = create_app()
```

- 실제 구현(`04_MVP/seelo/backend/seelo/main.py`)에서 더한 것(2026-10-27):
  - `request_id` 미들웨어가 접근 로그 `http_request`(method, path, status, ms)를 남긴다. 쿼리 문자열은 남기지 않는다(05 12장). 그래서 uvicorn은 `--no-access-log`.
  - `setup_logging(level, cache=APP_ENV != "test")`: 테스트에서는 로거를 캐시하지 않아 `capture_all_logs`가 로그를 모을 수 있다.
  - `APP_ENV`가 prod가 아니면 `allow_test_email_domain()`으로 `.test` 이메일을 허용한다(시드 계정, 17 5장).
  - `create_app(with_queue=False)`: 테스트용으로 작업 큐 연결 없이 앱을 만든다.
- 화면 타입 생성용 OpenAPI 파일: `scripts/export_openapi.py`가 `app.openapi()`를 `frontend/openapi.json`으로 저장한다(운영에서는 `/api/openapi.json`을 끈다).

## 7. `seelo/api/deps.py`

```python
import hmac
from datetime import timedelta
from fastapi import Depends, Request
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession
from seelo.db import get_db
from seelo.errors import AppError
from seelo.models.user import Session as SessionRow, User
from seelo.security.csrf import CSRF_COOKIE, CSRF_HEADER, verify_signature
from seelo.security.sessions import SESSION_COOKIE
from seelo.security.tokens import hash_token
from seelo.timeutil import now_utc

SAFE_METHODS = {"GET", "HEAD", "OPTIONS"}


async def load_session(request: Request, db: AsyncSession = Depends(get_db)) -> None:
    """모든 /api 요청의 첫 의존성. request.state.user·session_row를 채운다(없으면 None)."""
    request.state.user = None
    request.state.session_row = None
    token = request.cookies.get(SESSION_COOKIE)
    if not token:
        return
    row = await db.scalar(select(SessionRow).where(SessionRow.token_hash == hash_token(token)))
    if row is None:
        return
    if row.expires_at <= now_utc():
        await db.delete(row)
        await db.commit()
        return
    user = await db.get(User, row.user_id)
    if user is None or user.email_verified_at is None:
        return
    if row.last_seen_at < now_utc() - timedelta(minutes=5):
        row.last_seen_at = now_utc()
        await db.commit()
    request.state.user = user
    request.state.session_row = row


async def verify_csrf(request: Request) -> None:
    if request.method in SAFE_METHODS or request.url.path.startswith("/api/oauth/"):
        return
    header = request.headers.get(CSRF_HEADER)
    cookie = request.cookies.get(CSRF_COOKIE)
    if not header or not cookie or not hmac.compare_digest(header, cookie) or not verify_signature(cookie):
        raise AppError("CSRF_FAILED")
    row = request.state.session_row
    if row is not None and not hmac.compare_digest(hash_token(cookie), row.csrf_token_hash):
        raise AppError("CSRF_FAILED")


def require_user(request: Request) -> User:
    user: User | None = request.state.user
    if user is None:
        raise AppError("AUTH_REQUIRED")
    if user.terms_agreed_at is None:
        raise AppError("TERMS_REQUIRED")
    return user


def require_admin(user: User = Depends(require_user)) -> User:
    if user.role != "admin":
        raise AppError("FORBIDDEN")
    return user
```

```python
from seelo.security.tokens import new_token

_ph = PasswordHasher()                     # 기본값이 Argon2id
PASSWORD_RE = re.compile(r"(?=.*[A-Za-z])(?=.*\d).{8,128}")
DUMMY_HASH = _ph.hash(new_token())

    return _ph.hash(p)
```

```python
# csrf.py
import hmac, secrets
from seelo.config import get_settings

CSRF_COOKIE = "seelo_csrf"
CSRF_HEADER = "X-CSRF-Token"

def _sign(raw: str) -> str:
    key = get_settings().secret_key.get_secret_value().encode()
    return hmac.new(key, raw.encode(), "sha256").hexdigest()[:32]

def new_csrf_value() -> str:
    raw = secrets.token_urlsafe(24)
    return f"{raw}.{_sign(raw)}"

def verify_signature(value: str) -> bool:
    raw, sep, sig = value.rpartition(".")
    return bool(sep) and hmac.compare_digest(sig, _sign(raw))

def set_csrf_cookie(response, value: str) -> None:
    s = get_settings()
    response.set_cookie(CSRF_COOKIE, value, max_age=s.session_ttl_days * 86400,
                        httponly=False, secure=s.cookie_secure, samesite="lax", path="/")
```

```python
    """세션 행을 추가한다(커밋은 부르는 쪽). 세션·CSRF 쿠키를 응답에 단다."""
```

```python
def pk() -> Mapped[int]:
    return mapped_column(BigInteger, Identity(always=True), primary_key=True)
```

- 등록 예: `await jobs.defer(session, call_ai, args={"execution_id": e.id}, links=JobLinks(store_id=s, run_id=r), queueing_lock=f"callAi:{e.id}", queue=f"ai_{e.provider}")`(06 3장 `queue` 인자).

## 13. 메일 등록(`seelo/services/email_service.py`)

```python
from sqlalchemy.dialects.postgresql import insert
from seelo.config import get_settings
from seelo.models.ops import EmailLog
from seelo.security.crypto import encrypt_json
from seelo.services import jobs
from seelo.worker.tasks_email import send_email

async def queue(session, type_: str, *, user=None, to: str | None = None,
                params: dict | None = None, secret: dict | None = None,
                dedupe_key: str | None = None) -> int | None:
    """사건 트랜잭션 안에서 부른다. 커밋은 부르는 쪽.
    받는 사람: to(직접 지정, 시험 메일 등) → user.email → 관리자 주소(ADMIN_NOTIFY_EMAILS)."""
    to = to or (user.email if user is not None else ",".join(get_settings().admin_emails))
    if not to:
        return None
    stmt = (insert(EmailLog)
            .values(user_id=user.id if user else None, type=type_, to_email=to,
                    template_params=params or {},
                    secret_payload=encrypt_json(secret) if secret else None,
                    dedupe_key=dedupe_key, status="queued")
            .on_conflict_do_nothing(index_elements=["dedupe_key"])
            .returning(EmailLog.id))
    log_id = await session.scalar(stmt)
    if log_id is None:
        return None                                   # 같은 사건 메일이 이미 있음
    await jobs.defer(session, send_email, args={"email_log_id": log_id},
                     links=jobs.JobLinks(), queueing_lock=f"email:{log_id}")
    return log_id
```

## 14. API 경로 작성 예(`seelo/api/stores.py`)

```python
from fastapi import APIRouter, Depends
from sqlalchemy.ext.asyncio import AsyncSession
from seelo.api.deps import require_user
from seelo.db import get_db
from seelo.integrations import Clients, get_clients
from seelo.models.user import User
from seelo.schemas.store import IdentifyRequest, IdentifyResult, IdentifySelectRequest
from seelo.services import identify_service

router = APIRouter(prefix="/stores", tags=["stores"])


@router.post("/identify", response_model=IdentifyResult)
async def identify(body: IdentifyRequest, user: User = Depends(require_user),
                   db: AsyncSession = Depends(get_db), clients: Clients = Depends(get_clients)):
    return await identify_service.identify(db, clients, user, body)


@router.post("/identify/select", response_model=IdentifyResult)
async def identify_select(body: IdentifySelectRequest, user: User = Depends(require_user),
                          db: AsyncSession = Depends(get_db), clients: Clients = Depends(get_clients)):
    return await identify_service.select(db, clients, user, body)
```

- 모든 경로에 `response_model`을 단다(응답에 넣지 말아야 할 필드가 새지 않게, 05 11장).
- 경로 함수 안에서 DB 쿼리를 직접 쓰지 않는다. 서비스 함수가 커밋까지 하고 스키마 객체를 돌려준다.
